FROM node:20-alpine AS builder
WORKDIR /usr/src/app
COPY agent-cli/package*.json ./
RUN npm ci
COPY agent-cli/ ./
RUN npm run build

FROM node:20-alpine AS runner
WORKDIR /usr/src/app
ENV NODE_ENV=production
COPY agent-cli/package*.json ./
RUN npm ci --only=production
COPY --from=builder /usr/src/app/dist ./dist
EXPOSE 9090
USER node
CMD ["node", "dist/index.js"]

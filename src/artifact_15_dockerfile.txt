# Multi-stage production build
FROM node:20-alpine AS builder

WORKDIR /usr/src/app
COPY agent-daemon/package*.json ./
COPY agent-daemon/tsconfig.json ./
RUN npm ci

COPY agent-daemon/src ./src
RUN npm run build

FROM node:20-alpine AS runner
WORKDIR /app

ENV NODE_ENV=production
COPY agent-daemon/package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist
RUN mkdir -p /app/credentials

# Non-root user for enterprise runtime security
USER node

EXPOSE 8080
CMD ["node", "dist/run-agent.js"]
